
program         :=  { decl } EOF ;
decl            :=  importDecl | classDecl | funDecl | varDecl
importDecl      :=  "import" IDENT ";"
classDecl       :=  "class" IDENT [ ":" IDENT { "," IDENT } ] block
funDecl         :=  (["pub"] | ["priv"]) ["static"] IDENT "(" [ params ] ")" type block
varDecl         :=  type IDENT [ "=" expr ] ";"
params          :=  param { "," param }
param           :=  type IDENT

block           :=  "{" { stmt } "}"
stmt            :=  varDecl | exprStmt | ifStmt | whileStmt | returnStmt
exprStmt        :=  assignment | expr ";"
ifStmt          :=  "if" "(" expr ")" block [{"elseif" "(" expr ")" block }] ["else" block]
whileStmt       :=  "while" "(" expr ")" block
returnStmt      :=  "return" [ expr ] ";"

expr	        :=	logic_or
assignment      :=	IDENT "=" logic_or
logic_or        :=	logic_and { "||" logic_and }
logic_and       :=	equality { "&&" equality}
equality        :=	comparison { ("==" | "!=") comparison }
comparison      :=	term { ( "<" | ">" | "<=" | ">=" ) term }
term	        :=	factor { ("+" | "-") factor }
factor	        :=	unary { ("*" | "/") unary }
unary	        :=	[("!" | "-")] primary
TODO: For unary should we restrict the unary to specific primarys? For example - doesn't make sense for a boolean
primary	        :=	NUMBER | STRING | "true" | "false" | IDENT | funInvocation | paren
TODO: primitive?
primarySuffix   := "(" [primary { "," primary } ] ")" { memberAccess} | { memberAccess }
memberAccess    :=  ("." | "->") (funInvocation | IDENT)
paren           :=  "(" expr ")"

type	        :=	[const]  ("int32" | "float64" | "bool" | IDENT) (["*"] | ["&"])
